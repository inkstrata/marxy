const skip = !existsSync(webkit.executablePath()) && process.env.MARXY_BROWSER_TESTS_REQUIRED !== '1'
  ? 'Playwright WebKit is not installed here; MARXY_BROWSER_TESTS_REQUIRED=1 makes this a failure'
  : false;
const test = (name, fn) => nodeTest(name, { skip }, fn);

const repoRoot = new URL('../../../', import.meta.url).pathname;
const desktopRoot = new URL('..', import.meta.url).pathname;
const corpusDir = join(repoRoot, 'fixtures', 'corpus');
const outDir = mkdtempSync(join(tmpdir(), 'marxy-seams-'));
let server;
let base;

before(async () => {
  if (skip) return;
  await build({
    root: desktopRoot,
    logLevel: 'silent',
    build: {
      outDir,
      emptyOutDir: true,
      rollupOptions: {
        input: {
          app: join(desktopRoot, 'app.html'),
          paletteBoot: join(desktopRoot, 'test/palette-boot.html'),
        },
      },
    },
  });
  const types = { '.html': 'text/html', '.ttf': 'font/ttf', '.js': 'text/javascript', '.css': 'text/css' };
  server = createServer((req, res) => {
    const pathname = decodeURIComponent(new URL(req.url, 'http://x').pathname);
    const path = join(outDir, pathname.endsWith('/') ? `${pathname}index.html` : pathname);
    if (!existsSync(path)) { res.statusCode = 404; return res.end(); }
    res.setHeader('Content-Type', types[extname(path)] ?? 'application/octet-stream');
    res.end(readFileSync(path));
  });
  await new Promise((r) => server.listen(0, '127.0.0.1', r));
  base = `http://127.0.0.1:${server.address().port}/`;
});
after(() => server?.close());

function b64(path) {
  return readFileSync(path).toString('base64');
}

async function boot(page, file, argv) {
  const docPath = `/corpus/${file}`;
  await page.goto(`${base}test/palette-boot.html`);
  await page.waitForFunction(() => typeof window.marxyPaletteBoot?.start === 'function');
  await page.evaluate(async ({ files, argv }) => {
    window.__marxyOpsBoot = await window.marxyPaletteBoot.start(files, argv, []);
    window.__marxyOrigBytes = new Uint8Array(await window.__marxyOpsBoot.handle.shell.readFile(argv[0]));
  }, { files: { [docPath]: b64(join(corpusDir, file)) }, argv: [docPath] });
  await page.waitForFunction(() => typeof window.marxySelection?.getSelectionState === 'function');
  await page.waitForFunction(() => window.__marxyTasksReady === true);
  await page.waitForFunction(() => typeof window.marxyDocumentEdit === 'function');
  await page.waitForFunction(() => window.__marxyOpenSynced === true);
  return docPath;
}

function modKey(platform) {
  return platform === 'MacIntel' ? 'Meta' : 'Control';
}

const disk = (page, path) => page.evaluate(async (p) => new TextDecoder().decode(await window.__marxyOpsBoot.handle.shell.readFile(p)), path);
const settle = (page) => page.waitForTimeout(600);

// H5 (seams pass, MARXY-248): a failed write must not leave an undo entry that corrupts the next undo.
test('H5: undo after a failed save leaves the file as it was', async () => {
  const file = '03-ai-plan.md';
  const orig = readFileSync(join(corpusDir, file), 'utf8');
  const browser = await launchWebkit();
  try {
    const page = await browser.newPage({ viewport: { width: 960, height: 900 } });
    const docPath = await boot(page, file, [`/corpus/${file}`]);
    const failed = await page.evaluate(async () => {
      const shell = window.__marxyOpsBoot.handle.shell;
      const real = shell.writeFileAtomic;
      shell.writeFileAtomic = () => Promise.reject(new Error('disk full'));
      let err = null;
      try { await window.marxyHarnessAlignTable(); } catch (e) { err = String(e); }
      shell.writeFileAtomic = real;
      return err;
    });
    await settle(page);
    const afterFail = await disk(page, docPath);
    const mod = modKey(await page.evaluate(() => navigator.platform));
    await page.keyboard.press(`${mod}+KeyZ`);
    await settle(page);
    const afterUndo = await disk(page, docPath);
    console.log('H5 failed-write error:', failed);
    console.log('H5 disk unchanged after failed align:', afterFail === orig);
    console.log('H5 disk after undo === original:', afterUndo === orig, 'length', orig.length, '->', afterUndo.length);
    if (afterUndo !== orig) {
      const i = [...orig].findIndex((c, k) => c !== afterUndo[k]);
      console.log('H5 first difference at char', i, JSON.stringify(orig.slice(i - 40, i + 80)), '=>', JSON.stringify(afterUndo.slice(i - 40, i + 80)));
    }
    assert.equal(afterUndo, orig);
  } finally {
    await browser.close();
  }
});

// H6 (seams pass, MARXY-248): undo across a Rendered edit, a Source edit and back.
test('H6: undo walks back a Source edit, then a Rendered edit', async () => {
  const file = '03-ai-plan.md';
  const orig = readFileSync(join(corpusDir, file), 'utf8');
  const ast = parseMarkdown(readFileSync(join(corpusDir, file)), { file });
  const markers = [];
  const walk = (n) => { if (n.type === 'taskMarker') markers.push(n); for (const c of n.children ?? []) walk(c); };
  walk(ast);
  const browser = await launchWebkit();
  try {
    const page = await browser.newPage({ viewport: { width: 960, height: 900 } });
    const docPath = await boot(page, file, [`/corpus/${file}`]);
    const mod = modKey(await page.evaluate(() => navigator.platform));
    const log = [];
    const snap = async (step) => {
      const d = await disk(page, docPath);
      const buf = await page.evaluate(() => { const d = window.__marxyOpsBoot.handle.openDocument?.(); return d ? new TextDecoder().decode(d.buffer.bytes) : null; });
      log.push({ step, diskIsOrig: d === orig, diskHasTick: d !== orig && d.replace('[x]', '[ ]') !== d, diskHasSourceText: d.includes('SEAMS'), bufferHasSourceText: buf?.includes('SEAMS') ?? 'n/a', mode: await page.evaluate(() => document.documentElement.dataset.marxyMode ?? document.body.dataset.marxyMode ?? null) });
    };
    await page.evaluate((start) => document.querySelector(`#doc input[data-marxy-s="${start}"]`)?.dispatchEvent(new MouseEvent('click', { bubbles: true, cancelable: true })), markers[0].src.start);
    await settle(page); await snap('after Rendered toggle');
    await page.keyboard.press(`${mod}+KeyE`);
    await page.waitForSelector('.cm-content', { timeout: 5000 });
    await page.click('.cm-content');
    await page.keyboard.press(`${mod}+ArrowDown`);
    await page.keyboard.type('\nSEAMS\n');
    await settle(page); await snap('typed in Source');
    await page.keyboard.press(`${mod}+KeyE`);
    await settle(page); await snap('back in Rendered');
    await page.click('#doc', { position: { x: 5, y: 5 } }).catch(() => {});
    await page.keyboard.press(`${mod}+KeyZ`);
    await settle(page); await snap('undo 1');
    await page.keyboard.press(`${mod}+KeyZ`);
    await settle(page); await snap('undo 2');
    console.log('H6 trace:\n' + log.map((l) => JSON.stringify(l)).join('\n'));
    assert.equal(await disk(page, docPath), orig, 'two undos should return to the original bytes');
  } finally {
    await browser.close();
  }
});
